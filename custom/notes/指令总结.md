#### 运动

##### 寻找设备

```cmd
#端口
lerobot-find-port

#相机
lerobot-find-cameras opencv
```

##### 操纵

```cmd
lerobot-teleoperate ^
    --robot.type=so100_follower ^
    --robot.port=COM8 ^
    --robot.id=follower_arm1 ^
    --teleop.type=so100_leader ^
    --teleop.port=COM6 ^
    --teleop.id=leader_arm1 ^
    --display_data=true ^
    --robot.cameras="{robot1: {type: opencv, index_or_path: 0, width: 640, height: 480, fps: 30}, overall: {type: opencv, index_or_path: 1, width: 640, height: 480, fps: 30}}"
```

##### 录制

```cmd
lerobot-record ^
    --robot.type=so100_follower ^
    --robot.port=COM8 ^
    --robot.id=follower_arm1 ^
    --teleop.type=so100_leader ^
    --teleop.port=COM6 ^
    --teleop.id=leader_arm1 ^
    --display_data=true ^
    --robot.cameras="{robot1: {type: opencv, index_or_path: 0, width: 640, height: 480, fps: 30}, overall: {type: opencv, index_or_path: 1, width: 640, height: 480, fps: 30}}" ^
    --dataset.repo_id=Vacuame/grab_box ^
    --dataset.num_episodes=1 ^
    --dataset.single_task="Grab the white cube" ^
    --dataset.push_to_hub=false ^
    --resume=true
```

#### 训练

##### 训练

```cmd
lerobot-train ^
  --dataset.repo_id=Vacuame/triple_data ^
  --output_dir=outputs/train/my_policy3 ^
  --policy.repo_id=Vacuame/my_policy3 ^
  --job_name=customACT_so100test ^
  --policy.type=customACT ^
  --policy.device=cpu ^
  --policy.push_to_hub=False ^
  --wandb.enable=false ^
  --steps=10 ^
  --eval_freq=2 ^
  --log_freq=1
```
